cmake_minimum_required(VERSION 3.16)
project(telemetry_batch VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(telemetry STATIC telemetry.c)
target_include_directories(telemetry PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(telemetry_send sender.c)
target_link_libraries(telemetry_send PRIVATE telemetry)
if(WIN32)
  target_link_libraries(telemetry_send PRIVATE winhttp)
else()
  find_package(Threads REQUIRED)
  find_package(CURL REQUIRED)
  target_link_libraries(telemetry_send PRIVATE Threads::Threads CURL::libcurl m)
endif()
foreach(target telemetry telemetry_send)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_telemetry test_telemetry.c)
  target_link_libraries(test_telemetry PRIVATE telemetry)
  if(NOT WIN32)
    target_link_libraries(test_telemetry PRIVATE m)
  endif()
  if(MSVC)
    target_compile_options(test_telemetry PRIVATE /W4)
  else()
    target_compile_options(test_telemetry PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME core COMMAND test_telemetry)
endif()
include(GNUInstallDirs)
install(TARGETS telemetry_send RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS telemetry ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES telemetry.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/telemetry_batch)
install(FILES README.md REVIEW.md example.csv DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CPACK_PACKAGE_NAME telemetry-batch)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "telemetry-batch-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
if(WIN32)
  set(CPACK_GENERATOR ZIP)
else()
  set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
