cmake_minimum_required(VERSION 3.16)
project(telemetry_batch VERSION 0.3.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(cmake/Architecture.cmake)
add_library(telemetry STATIC telemetry.c time_sync.c)
target_include_directories(telemetry PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(telemetry_ui STATIC config.c monitor.c)
target_link_libraries(telemetry_ui PUBLIC telemetry)
add_executable(telemetry_send sender.c)
add_executable(telemetry_monitor sender.c)
target_compile_definitions(telemetry_monitor PRIVATE TB_MONITOR_DEFAULT=1)
if(NOT WIN32)
  find_package(Threads REQUIRED)
  find_package(CURL REQUIRED)
endif()
foreach(target telemetry_send telemetry_monitor)
  target_link_libraries(${target} PRIVATE telemetry telemetry_ui)
  if(WIN32)
    target_link_libraries(${target} PRIVATE winhttp shell32)
  else()
    target_link_libraries(${target} PRIVATE Threads::Threads CURL::libcurl m)
  endif()
endforeach()
foreach(target telemetry telemetry_ui telemetry_send telemetry_monitor)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_telemetry test_telemetry.c)
  target_link_libraries(test_telemetry PRIVATE telemetry)
  add_executable(test_time_sync test_time_sync.c)
  target_link_libraries(test_time_sync PRIVATE telemetry)
  add_executable(test_config test_config.c config.c)
  foreach(target test_telemetry test_time_sync test_config)
    if(NOT WIN32)
      target_link_libraries(${target} PRIVATE m)
    endif()
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endforeach()
  add_test(NAME core COMMAND test_telemetry)
  add_test(NAME clock COMMAND test_time_sync)
  add_test(NAME config COMMAND test_config)
endif()
include(GNUInstallDirs)
install(TARGETS telemetry_send telemetry_monitor RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS telemetry ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES telemetry.h time_sync.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/telemetry_batch)
install(FILES README.md REVIEW.md example.csv telemetry.ini.example DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CPACK_PACKAGE_NAME telemetry-batch)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_STRIP_FILES TRUE)
set(CPACK_PACKAGE_FILE_NAME "telemetry-batch-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${TB_TARGET_ARCH}")
if(WIN32)
  set(CPACK_GENERATOR ZIP)
else()
  set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
