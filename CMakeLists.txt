cmake_minimum_required(VERSION 3.21)
project(mavlink_telemetry_data_router VERSION 0.5.0 LANGUAGES C)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build: cmake --preset windows-x64 or linux-x64")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
# Each preset has one isolated build tree containing both application variants.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
include(cmake/Architecture.cmake)
include(GNUInstallDirs)
# Developer tests are opt-in; ordinary builds produce only the two programs.
option(BUILD_TESTING "Build developer verification programs" OFF)
include(CTest)

function(tb_strict_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
add_subdirectory(src)

install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES src/vendor/MAVLINK_COPYING.txt DESTINATION ${CMAKE_INSTALL_DOCDIR}/licenses)
install(FILES docs/REFERENCE.md docs/REVIEW.md DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
install(FILES examples/example.csv DESTINATION ${CMAKE_INSTALL_DOCDIR}/examples)
install(FILES config/telemetry.ini.example DESTINATION ${CMAKE_INSTALL_DOCDIR}/config)
set(CPACK_PACKAGE_NAME mavlink_telemetry_data_router)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_STRIP_FILES TRUE)
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_SOURCE_DIR}/build/packages")
set(CPACK_PACKAGE_FILE_NAME "mavlink_telemetry_data_router-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${TB_TARGET_ARCH}")
if(WIN32)
  set(CPACK_GENERATOR ZIP)
else()
  set(CPACK_GENERATOR TGZ)
endif()
set(CPACK_SOURCE_GENERATOR "ZIP;TGZ")
set(CPACK_SOURCE_PACKAGE_FILE_NAME "mavlink_telemetry_data_router-${PROJECT_VERSION}-source")
set(CPACK_SOURCE_IGNORE_FILES "/\\.git/;/\\.build[^/]*/;/build[^/]*/;/dist/;/install/;/stage/;/__pycache__/;/CMakeUserPresets\\.json$;/telemetry\\.ini$;\\.ini\\.tmp\\.;\\.pyc$;\\.exe$;\\.zip$;\\.tar\\.gz$")
include(CPack)
