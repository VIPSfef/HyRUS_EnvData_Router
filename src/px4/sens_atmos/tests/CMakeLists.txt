cmake_minimum_required(VERSION 3.16)
project(sens_atmos_stream_tests LANGUAGES CXX)
include(CTest)
add_executable(test_sens_atmos test_sens_atmos.cpp ../SENS_ATMOS.hpp)
target_compile_features(test_sens_atmos PRIVATE cxx_std_14)
target_include_directories(test_sens_atmos PRIVATE stubs ..)
if(MSVC)
    target_compile_options(test_sens_atmos PRIVATE /W4 /WX)
else()
    target_compile_options(test_sens_atmos PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME sens_atmos_stream COMMAND test_sens_atmos)
