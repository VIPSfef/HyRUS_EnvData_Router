cmake_minimum_required(VERSION 3.16)
project(shtc3_probe_tests LANGUAGES CXX)
include(CTest)
add_executable(test_shtc3_probe test_shtc3_probe.cpp)
target_compile_features(test_shtc3_probe PRIVATE cxx_std_14)
target_include_directories(test_shtc3_probe PRIVATE .. ../../shtc3)
if(MSVC)
    target_compile_options(test_shtc3_probe PRIVATE /W4 /WX)
else()
    target_compile_options(test_shtc3_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME shtc3_probe COMMAND test_shtc3_probe)
