cmake_minimum_required(VERSION 3.21)
project(shtc3_core_tests LANGUAGES CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Build outside the source tree, e.g. build/shtc3-tests-windows or build/shtc3-tests-linux")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(test_shtc3 test_shtc3.cpp ../SHTC3Protocol.hpp)
if(MSVC)
  target_compile_options(test_shtc3 PRIVATE /W4 /WX)
else()
  target_compile_options(test_shtc3 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME shtc3_core COMMAND test_shtc3)
