add_library(telemetry STATIC telemetry.c time_sync.c weather.c)
target_include_directories(telemetry PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(telemetry_ui STATIC config.c monitor.c)
add_library(router_input STATIC router_input.c input_transport.c mavlink_input.c)
target_link_libraries(router_input PUBLIC telemetry)
target_link_libraries(telemetry_ui PUBLIC telemetry)

# Both are complete receiver/relay programs. Only the default screen differs.
add_executable(mavlink_telemetry_data_router_headless sender.c)
add_executable(mavlink_telemetry_data_router_interactive sender.c)
target_compile_definitions(mavlink_telemetry_data_router_interactive PRIVATE TB_MONITOR_DEFAULT=1)
if(NOT WIN32)
  find_package(Threads REQUIRED)
  find_package(CURL REQUIRED)
  target_link_libraries(router_input PRIVATE CURL::libcurl)
endif()
foreach(target mavlink_telemetry_data_router_headless mavlink_telemetry_data_router_interactive)
  target_link_libraries(${target} PRIVATE telemetry telemetry_ui router_input)
  if(WIN32)
    target_link_libraries(${target} PRIVATE winhttp shell32 bcrypt ws2_32)
  else()
    target_link_libraries(${target} PRIVATE Threads::Threads CURL::libcurl m)
  endif()
endforeach()
foreach(target telemetry telemetry_ui router_input mavlink_telemetry_data_router_headless mavlink_telemetry_data_router_interactive)
  tb_strict_warnings(${target})
endforeach()
# Optional demonstration endpoint, kept separate from the four receiver apps.
if(BUILD_TEST_SERVER)
  add_executable(mavlink_telemetry_data_router_test_server test_server.c)
  set_target_properties(mavlink_telemetry_data_router_test_server PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(mavlink_telemetry_data_router_test_server PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/tools")
  endforeach()
  if(WIN32)
    target_link_libraries(mavlink_telemetry_data_router_test_server PRIVATE ws2_32)
  else()
    target_link_libraries(mavlink_telemetry_data_router_test_server PRIVATE m)
  endif()
  tb_strict_warnings(mavlink_telemetry_data_router_test_server)
  install(TARGETS mavlink_telemetry_data_router_test_server RUNTIME DESTINATION tools)
endif()

if(BUILD_TESTING)
  add_subdirectory(tests)
endif()

install(TARGETS mavlink_telemetry_data_router_headless mavlink_telemetry_data_router_interactive RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS telemetry ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES telemetry.h time_sync.h weather.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/${PROJECT_NAME})
