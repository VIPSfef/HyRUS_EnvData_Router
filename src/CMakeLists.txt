if(NOT WIN32)
  find_package(Threads REQUIRED)
  find_package(CURL REQUIRED)
  target_link_libraries(router_input PRIVATE CURL::libcurl)
endif()
function(tb_link_router_runtime target)
  target_link_libraries(${target} PRIVATE telemetry telemetry_ui router_input udp_backup memory_queue)
  if(WIN32)
    target_link_libraries(${target} PRIVATE winhttp shell32 bcrypt ws2_32)
  else()
    target_link_libraries(${target} PRIVATE Threads::Threads CURL::libcurl m)
  endif()
endfunction()
foreach(target mavlink_telemetry_data_router_headless mavlink_telemetry_data_router_interactive)
  tb_link_router_runtime(${target})
endforeach()
foreach(target telemetry telemetry_ui qgc_bridge mavlink_rates udp_backup memory_queue router_input mavlink_telemetry_data_router_headless mavlink_telemetry_data_router_interactive)
  tb_strict_warnings(${target})
endforeach()
